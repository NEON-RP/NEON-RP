const NEON_API = 'https://neon-rp.fadehost.app';

(async function finishDiscordLogin() {
  const url = new URL(window.location.href);
  const ticket = url.searchParams.get('discord_auth_ticket');
  const loginError = url.searchParams.get('discord_login');

  if (loginError) {
    const messages = {
      not_configured: 'تسجيل دخول ديسكورد غير مكتمل الإعداد على الاستضافة.',
      state_error: 'تعذر التحقق من جلسة تسجيل الدخول. حاول مرة أخرى.',
      cancelled: 'تم إلغاء تسجيل الدخول بديسكورد.',
      failed: 'فشل تسجيل الدخول بديسكورد. حاول مرة أخرى.'
    };
    // Replace this with the website's existing status/toast element if available.
    console.warn(messages[loginError] || 'تعذر تسجيل الدخول بديسكورد.');
    url.searchParams.delete('discord_login');
    history.replaceState({}, '', url.toString());
    return;
  }
  if (!ticket) return;

  try {
    const response = await fetch(`${NEON_API}/api/auth/discord/session`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ ticket })
    });
    const data = await response.json();
    if (!response.ok || !data.ok || !data.user) throw new Error(data.error || 'Login failed');

    // Store only the public profile; NEVER store an OAuth access token.
    localStorage.setItem('neonDiscordUser', JSON.stringify(data.user));

    // Prefill the existing application fields if these IDs are present.
    const idField = document.querySelector('#discordId, [name="discordId"]');
    const usernameField = document.querySelector('#discordUsername, [name="discordUsername"]');
    if (idField) idField.value = data.user.id;
    if (usernameField) usernameField.value = data.user.username;

    // Update an existing login label, if the page has one.
    const loginLabel = document.querySelector('#discordLoginStatus, [data-discord-login-status]');
    if (loginLabel) loginLabel.textContent = `تم ربط حساب ديسكورد: ${data.user.username}`;
  } catch (error) {
    console.error('NEON RP Discord login:', error);
    const loginLabel = document.querySelector('#discordLoginStatus, [data-discord-login-status]');
    if (loginLabel) loginLabel.textContent = 'فشل ربط حساب ديسكورد. اضغط تسجيل الدخول مرة أخرى.';
  } finally {
    url.searchParams.delete('discord_auth_ticket');
    history.replaceState({}, '', url.toString());
  }
})();
